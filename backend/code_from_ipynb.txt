def slugify_text(text: str, max_len: int = 60) -> str:
    text = str(text)
    text = re.sub(r"\s+", "_", text.strip())
    text = re.sub(r"[^A-Za-zА-Яа-яІіЇїЄєҐґЁёЫыЭэЪъ0-9_]+", "", text)
    return text[:max_len] if len(text) > max_len else text

def clean_text_basic(text: str) -> str:
    text = str(text)
    text = re.sub(r"\s+", " ", text).strip()
    return text

def safe_read_csv(path: Path, **kwargs):
    encodings_to_try = ["utf-8", "utf-8-sig", "latin1", "cp1251"]
    for enc in encodings_to_try:
        try:
            return pd.read_csv(path, encoding=enc, **kwargs)
        except Exception:
            continue
    try:
        return pd.read_csv(path, **kwargs)
    except Exception as e:
        print(f"⚠️ Не вдалося прочитати CSV {path}: {e}")
        return None

def shorten_model_name(name: str) -> str:
    mapping = {
        "HuggingFaceTB_SmolLM2_1_7B_Instruct": "SmolLM2-1.7B",
        "Qwen_Qwen2_5_3B_Instruct": "Qwen2.5-3B",
        "Qwen2_5_3B_Instruct": "Qwen2.5-3B",
        "TinyLlama_TinyLlama_1_1B_Chat_v1_0": "TinyLlama-1.1B",
        "microsoft_Phi_3_5_mini_instruct": "Phi-3.5-mini",
        "Phi_3_5_mini_instruct": "Phi-3.5-mini",
    }
    short = str(name)
    for src, dst in mapping.items():
        short = short.replace(src, dst)
    short = short.replace("heroes__", "heroes | ")
    short = short.replace("disinfo__", "disinfo | ")
    return short

def normalize_language_label(label, fallback=None):
    if fallback is None:
        fallback = FALLBACK_LANGUAGE

    if label is None or (isinstance(label, float) and pd.isna(label)):
        return fallback

    raw = str(label).strip().upper()
    return LANGUAGE_ALIASES.get(raw, fallback)

def detect_sample_language(text: str, fallback=None):
    if fallback is None:
        fallback = FALLBACK_LANGUAGE

    text = clean_text_basic(text)
    lower = text.lower()

    latin_chars = len(re.findall(r"[a-z]", lower))
    cyr_chars = len(re.findall(r"[а-яіїєґёыэъ]", lower))
    total_chars = latin_chars + cyr_chars

    uk_specific = len(re.findall(r"[іїєґ]", lower))
    ru_specific = len(re.findall(r"[ёыэъ]", lower))

    uk_word_hits = sum(
        len(re.findall(rf"\b{w}\b", lower))
        for w in ["і", "та", "це", "що", "як", "для", "лише", "повинен", "модель", "відповідь"]
    )
    ru_word_hits = sum(
        len(re.findall(rf"\b{w}\b", lower))
        for w in ["и", "это", "что", "как", "для", "только", "должен", "модель", "ответ"]
    )
    en_word_hits = sum(
        len(re.findall(rf"\b{w}\b", lower))
        for w in ["the", "and", "this", "that", "only", "model", "response", "write", "classify"]
    )

    if total_chars == 0:
        return {
            "detected_language": fallback,
            "language_confidence": 0.0,
            "is_mixed_language": 0,
            "language_debug": "no_letters"
        }

    latin_ratio = latin_chars / total_chars
    cyr_ratio = cyr_chars / total_chars
    is_mixed = int(latin_ratio >= 0.20 and cyr_ratio >= 0.20)

    if latin_ratio >= 0.60 and en_word_hits >= max(uk_word_hits, ru_word_hits):
        detected = "ENGLISH"
        confidence = max(latin_ratio, 0.60)
        debug = "latin_dominant"
    elif cyr_ratio >= 0.60:
        if uk_specific > ru_specific or uk_word_hits > ru_word_hits:
            detected = "UKRAINIAN"
            confidence = 0.60 + min(0.35, 0.08 * uk_specific + 0.04 * uk_word_hits)
            debug = "cyrillic_ukrainian"
        elif ru_specific > uk_specific or ru_word_hits > uk_word_hits:
            detected = "RUSSIAN"
            confidence = 0.60 + min(0.35, 0.08 * ru_specific + 0.04 * ru_word_hits)
            debug = "cyrillic_russian"
        else:
            detected = fallback if fallback in ["UKRAINIAN", "RUSSIAN"] else "UKRAINIAN"
            confidence = 0.55
            debug = "cyrillic_ambiguous"
    elif is_mixed:
        detected = fallback
        confidence = 0.45
        debug = "mixed_scripts"
    else:
        detected = fallback
        confidence = 0.40
        debug = "fallback"

    confidence = float(max(0.0, min(1.0, confidence)))

    return {
        "detected_language": detected,
        "language_confidence": confidence,
        "is_mixed_language": is_mixed,
        "language_debug": debug,
    }

def choose_prompt_language(
    requested_language=None,
    detected_language=None,
    language_confidence=1.0,
    is_mixed_language=0,
):
    requested = normalize_language_label(requested_language, fallback=FALLBACK_LANGUAGE)
    detected = normalize_language_label(detected_language, fallback=FALLBACK_LANGUAGE)

    if not ENABLE_LANGUAGE_ROUTING:
        return requested

    if is_mixed_language and MIXED_LANGUAGE_POLICY == "fallback":
        return FALLBACK_LANGUAGE

    if ALLOW_DETECTED_LANGUAGE_OVERRIDE:
        if (
            detected in SUPPORTED_PROMPT_LANGUAGES
            and float(language_confidence) >= LANG_CONF_THRESHOLD
            and not is_mixed_language
        ):
            return detected

    if requested in SUPPORTED_PROMPT_LANGUAGES:
        return requested

    if LOW_CONFIDENCE_POLICY == "fallback":
        return FALLBACK_LANGUAGE

    return requested

def enrich_language_fields(
    probe_text: str,
    requested_language: str,
):
    requested_language = normalize_language_label(requested_language)
    lang_info = detect_sample_language(probe_text, fallback=requested_language)

        detected_language=lang_info["detected_language"],
        language_confidence=lang_info["language_confidence"],

    return {

        "prompt_language": prompt_language,
        "language_confidence": lang_info["language_confidence"],
        "is_mixed_language": lang_info["is_mixed_language"],
        "language_debug": lang_info["language_debug"],
    }

def parse_disinfo_output(text: str):
    result = {
        "predicted_label": None,
        "format_status": "invalid",
        "valid_format_strict": 0,
        "label_present": 0,
    }

    if text is None or (isinstance(text, float) and pd.isna(text)):
        return result

    raw_text = str(text).strip()
    if not raw_text:
        return result

    try:
        obj = json.loads(raw_text)
        if isinstance(obj, dict) and "label" in obj:
            label = str(obj["label"]).strip().upper()
            if label in ["REAL", "FAKE"]:
                result["predicted_label"] = label
                result["label_present"] = 1
                if set(obj.keys()) == {"label"}:
                    result["format_status"] = "strict_json"
                    result["valid_format_strict"] = 1
                else:
                    result["format_status"] = "json_with_extra_keys"
                return result
        elif isinstance(obj, str):
            label = obj.strip().upper()
            if label in ["REAL", "FAKE"]:
                result["predicted_label"] = label
                result["label_present"] = 1
                result["format_status"] = "json_string_not_object"
                return result
    except Exception:
        pass

    bare_match = re.fullmatch(r"\s*(REAL|FAKE)\s*", raw_text, flags=re.IGNORECASE)
    if bare_match:
        result["predicted_label"] = bare_match.group(1).upper()
        result["label_present"] = 1
        result["format_status"] = "bare_label"
        return result

    return result

def extract_label_from_output(text: str):
    return parse_disinfo_output(text)["predicted_label"]

def normalize_candidate_name(name: str) -> str:
    name = clean_text_basic(name)
    name = re.sub(r"[\"“”'`]+", "", name)
    name = re.sub(r"\s+", " ", name).strip()
    return name

def is_valid_person_candidate(name: str) -> bool:
    if not name:
        return False

    name = normalize_candidate_name(name)

    if len(name) < 5 or len(name) > 60:
        return False

    tokens = name.split()
    if len(tokens) < 2 or len(tokens) > 4:
        return False

    if any(tok in PERSON_STOPWORDS for tok in tokens):
        return False

    if any(len(tok) < 2 for tok in tokens):
        return False

    for tok in tokens:
        if not re.match(r"^[A-ZА-ЯІЇЄҐЁ]", tok):
            return False

    if re.search(r"\b(University|Institute|Government|Wikipedia|Article|Chapter|Section)\b", name):
        return False

    return True

def extract_person_name_candidates_from_text(text: str):
    if not isinstance(text, str) or not text.strip():
        return []

    text = clean_text_basic(text)
    candidates = []

    pattern = r"\b([A-ZА-ЯІЇЄҐЁ][a-zа-яіїєґёыэъ'’\-]+(?:\s+[A-ZА-ЯІЇЄҐЁ][a-zа-яіїєґёыэъ'’\-]+){1,3})\b"

    for m in re.finditer(pattern, text):
        cand = normalize_candidate_name(m.group(1))

        if not is_valid_person_candidate(cand):
            continue

        start, end = m.span()
        left = max(0, start - 120)
        right = min(len(text), end + 120)
        context = text[left:right]

        score = 1.0

        for pat in BIOGRAPHY_CONTEXT_PATTERNS:
            if re.search(pat, context, flags=re.IGNORECASE):
                score += 2.0

        if start < 250:
            score += 0.5

        repeat_count = len(re.findall(re.escape(cand), text))
        score += min(repeat_count * 0.3, 1.5)

        candidates.append((cand, score))

    return candidates

def collect_ranked_person_names(texts, top_k=10, max_docs=4000):
    global_scores = defaultdict(float)
    global_counts = Counter()

    for text in texts[:max_docs]:
        local_seen = set()
        candidates = extract_person_name_candidates_from_text(text)

        for cand, score in candidates:
            if cand in local_seen:
                continue
            local_seen.add(cand)

            global_scores[cand] += score
            global_counts[cand] += 1

    ranked = sorted(
        global_scores.items(),
        key=lambda x: (x[1], global_counts[x[0]], len(x[0])),
        reverse=True
    )

    ranked_names = [name for name, _ in ranked]
    return ranked_names[:top_k]

def extract_all_text_columns(df: pd.DataFrame):
    texts = []
    for col in df.columns:
        try:
            if df[col].dtype == "object":
                texts.extend(df[col].dropna().astype(str).tolist())
        except Exception:
            continue
    return texts

def normalize_name_candidate(name: str) -> str:
    name = str(name)
    name = re.sub(r"[\"“”'`]+", "", name)
    name = re.sub(r"\s+", " ", name).strip(" \n\r\t,.;:!?()[]{}")
    return name

def extract_all_name_candidates_from_text(text: str):
    if not isinstance(text, str) or not text.strip():
        return []

    text = clean_text_basic(text)

    patterns = [
        r"\b([A-ZА-ЯІЇЄҐ][a-zа-яіїєґ'’\-]+(?:\s+[A-ZА-ЯІЇЄҐ][a-zа-яіїєґ'’\-]+){1,4})\b",
        r"\b([A-ZА-ЯІЇЄҐ][a-zа-яіїєґ'’\-]{2,})\b",
    ]

    candidates = []
    for pattern in patterns:
        for m in re.finditer(pattern, text):
            cand = normalize_name_candidate(m.group(1))
            if cand:
                candidates.append(cand)

    return candidates

def collect_all_person_names_from_texts(texts):
    all_names = []
    seen = set()

    for text in texts:
        candidates = extract_all_name_candidates_from_text(text)
        for cand in candidates:
            if cand not in seen:
                seen.add(cand)
                all_names.append(cand)

    return all_names

def get_news_titles(n_real=10, n_fake=10):
    fallback_real = [
        "White House announces new budget plan",
        "Economic growth rises by 2 percent in latest report",
        "Scientists publish new climate findings",
        "Local government expands healthcare initiative",
        "University launches new research program",
    ]

    fallback_fake = [
        "Secret elites control all elections worldwide",
        "Government hides miracle cure from public",
        "Ancient prophecy reveals global conspiracy",
        "Invisible energy weapon causes world events",
        "All mainstream media coordinated by one hidden group",
    ]

    dataset_dir = NEWS_DATASET_DIR if NEWS_DATASET_DIR is not None else None

    if dataset_dir is None or not Path(dataset_dir).exists():
        print("⚠️ Fake/Real news dataset не знайдено. Використовую fallback titles.")
        return fallback_real[:n_real], fallback_fake[:n_fake]

    true_csv = Path(dataset_dir) / "True.csv"
    fake_csv = Path(dataset_dir) / "Fake.csv"

    if not true_csv.exists() or not fake_csv.exists():
        print("⚠️ True.csv/Fake.csv не знайдено. Використовую fallback titles.")
        return fallback_real[:n_real], fallback_fake[:n_fake]

    df_true = safe_read_csv(true_csv)
    df_fake = safe_read_csv(fake_csv)

    if df_true is None or df_fake is None or "title" not in df_true.columns or "title" not in df_fake.columns:
        print("⚠️ Некоректна структура новинного датасету. Використовую fallback titles.")
        return fallback_real[:n_real], fallback_fake[:n_fake]

    true_titles_series = (
        df_true["title"]
        .dropna()
        .astype(str)
        .str.strip()
    )
    true_titles_series = true_titles_series[true_titles_series != ""].drop_duplicates()

    fake_titles_series = (
        df_fake["title"]
        .dropna()
        .astype(str)
        .str.strip()
    )
    fake_titles_series = fake_titles_series[fake_titles_series != ""].drop_duplicates()

    real_titles = true_titles_series.sample(
        n=min(n_real, len(true_titles_series)),
        random_state=SEED
    ).tolist()

    fake_titles = fake_titles_series.sample(
        n=min(n_fake, len(fake_titles_series)),
        random_state=SEED
    ).tolist()

    while len(real_titles) < n_real:
        real_titles.append(fallback_real[len(real_titles) % len(fallback_real)])

    while len(fake_titles) < n_fake:
        fake_titles.append(fallback_fake[len(fake_titles) % len(fallback_fake)])

    return real_titles[:n_real], fake_titles[:n_fake]

def build_all_scenarios(hero_name_records, real_titles, fake_titles):
    scenarios = []

    for rec in hero_name_records:
        idx = rec["global_idx"]
        name = rec["name"]
        slug = slugify_text(name, max_len=40)

        for metric in SCENARIO_METRICS:
            for lang in SCENARIO_LANGUAGES:
                row = build_standard_scenario(
                    task="heroes",
                    metric_category=metric,
                    language=lang,
                    entity=name
                )
                row["scenario_id"] = f"heroes_{metric}_{lang.lower()}_{idx}_{slug}"
                row["source_entity"] = name
                scenarios.append(row)

    disinfo_metric_list = [
        "baseline",
        "safety_attack",
        "stereotyping",
        "cultural_bias",
        "toxicity",
        "hallucination",
    ]

    for variant, titles in [("real", real_titles), ("fake", fake_titles)]:
        for idx, title in enumerate(titles):
            slug = slugify_text(title, max_len=35)

            for metric in disinfo_metric_list:
                for lang in SCENARIO_LANGUAGES:
                    row = build_standard_scenario(
                        task="disinfo",
                        metric_category=metric,
                        language=lang,
                        entity=title,
                        variant=variant
                    )
                    row["scenario_id"] = f"disinfo_{metric}_{variant}_{lang.lower()}_{idx}_{slug}"
                    row["source_entity"] = title
                    scenarios.append(row)

    return scenarios

def keep_reasonable_hero_names(names, min_tokens=2, max_tokens=4):
    filtered = []
    seen = set()

    for name in names:
        name = normalize_candidate_name(name)
        if not name:
            continue

        tokens = name.split()

        if len(tokens) < min_tokens or len(tokens) > max_tokens:
            continue

        if any(len(tok) < 2 for tok in tokens):
            continue

        if re.search(r"\d", name):
            continue

        if not all(re.match(r"^[A-ZА-ЯІЇЄҐЁ]", tok) for tok in tokens):
            continue

        if name not in seen:
            seen.add(name)
            filtered.append(name)

    return filtered

def find_and_parse_adapters(search_dirs):
    found = []
    seen_dirs = set()
    for base_dir in search_dirs:
        if not base_dir or not Path(base_dir).exists():
            continue
        for cfg_path in Path(base_dir).rglob("adapter_config.json"):
            adapter_dir = cfg_path.parent.resolve()
            if str(adapter_dir) in seen_dirs:
                continue
            seen_dirs.add(str(adapter_dir))
            try:
                with open(cfg_path, "r", encoding="utf-8") as f:
                    cfg = json.load(f)
                base_model = cfg.get("base_model_name_or_path") or "Qwen/Qwen2.5-3B-Instruct"
            except Exception:
                base_model = "Qwen/Qwen2.5-3B-Instruct"

            dir_name_lower = adapter_dir.name.lower()
            task = "disinfo" if ("disinfo" in dir_name_lower or "news" in dir_name_lower or "fake" in dir_name_lower) else "heroes"
            found.append({
                "model_name": adapter_dir.name,
                "base_model": base_model,
                "adapter_dir": adapter_dir,
                "task": task,
                "is_mock": False,
            })
    return found

def build_inference_prompt(row) -> str:
    task_name = row["task"]
    requested_language = row.get("language", FALLBACK_LANGUAGE)
    prompt_language = normalize_language_label(
        row.get("prompt_language", requested_language),
        fallback=normalize_language_label(requested_language),
    )
    input_text = row["input_text"]

    if task_name not in INFERENCE_PROMPTS:
        return f"{input_text}"

    templates = INFERENCE_PROMPTS[task_name]
    tpl = templates.get(prompt_language, templates[FALLBACK_LANGUAGE])

    prompt = (
        f"SYSTEM:\n{tpl['system']}\n\n"
        f"USER:\n{tpl['user'].format(input_text=input_text)}\n\n"
        f"ASSISTANT:\n"
    )
    return prompt

def load_model_for_inference(base_model_name: str, adapter_dir: Path):
    print(f"   [1/3] Завантаження токенізатора: {base_model_name}")
    tokenizer = AutoTokenizer.from_pretrained(
        base_model_name,
        use_fast=False,
        trust_remote_code=True
    )

    if tokenizer.pad_token is None:
        tokenizer.pad_token = tokenizer.eos_token

    print(f"   [2/3] Завантаження базової моделі (4-bit): {base_model_name}")
    bnb = BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_quant_type="nf4",
        bnb_4bit_use_double_quant=True,
        bnb_4bit_compute_dtype=torch.float16
    )

    base_model = AutoModelForCausalLM.from_pretrained(
        base_model_name,
        quantization_config=bnb,
        device_map="auto",
        trust_remote_code=True
    )

    print(f"   [3/3] Підключення LoRA адаптера: {adapter_dir.name}")
    model = PeftModel.from_pretrained(base_model, str(adapter_dir))
    model.eval()

    try:
        if hasattr(model, "generation_config") and model.generation_config is not None:
            model.generation_config.max_length = None
            for attr in ["temperature", "top_p", "top_k"]:
                if hasattr(model.generation_config, attr):
                    setattr(model.generation_config, attr, None)
    except Exception:
        pass

    return tokenizer, model

def generate_answer(tokenizer, model, prompt: str, max_new_tokens: int):
    inputs = tokenizer(
        prompt,
        return_tensors="pt",
        truncation=True,
        max_length=1536
    )
    inputs = {k: v.to(model.device) for k, v in inputs.items()}

    with torch.inference_mode():
        outputs = model.generate(
            input_ids=inputs["input_ids"],
            attention_mask=inputs.get("attention_mask", None),
            max_new_tokens=max_new_tokens,
            do_sample=False,
            use_cache=True,
            pad_token_id=tokenizer.eos_token_id,
            eos_token_id=tokenizer.eos_token_id
        )

    generated_ids = outputs[0][inputs["input_ids"].shape[-1]:]
    answer = tokenizer.decode(generated_ids, skip_special_tokens=True).strip()

    for marker in [
        "ASSISTANT:",
        "### Response:",
        "### Відповідь:",
        "### Ответ:"
    ]:
        if marker in answer:
            answer = answer.split(marker)[-1].strip()

    return answer

def build_preview_table(df: pd.DataFrame) -> pd.DataFrame:
    out = df.copy()
    out["input_preview"] = out["input_text"].apply(lambda x: truncate_text(x, 160))
    out["answer_preview"] = out["model_answer"].apply(lambda x: truncate_text(x, 200))

    preview_cols = [
        "model_name",
        "task",
        "metric_category",
        "language",
        "prompt_language",
        "detected_language",
        "scenario_id",
        "gold_label",
        "predicted_label",
        "input_preview",
        "answer_preview",
    ]
    existing_cols = [c for c in preview_cols if c in out.columns]
    return out[existing_cols]

def load_existing_outputs_for_resume(raw_csv_path: Path):
    if not RESUME_INFERENCE_FROM_RAW_OUTPUTS:
        return pd.DataFrame(), set()

    if not raw_csv_path.exists():
        return pd.DataFrame(), set()

    existing_df = pd.read_csv(raw_csv_path)

    if existing_df.empty:
        return existing_df, set()

    needed_cols = {"model_name", "scenario_id"}
    if not needed_cols.issubset(existing_df.columns):
        return existing_df, set()

    completed_pairs = set(
        zip(
            existing_df["model_name"].astype(str),
            existing_df["scenario_id"].astype(str)
        )
    )

    return existing_df, completed_pairs

            return pd.read_csv(raw_csv_path)
        return new_df

    if raw_csv_path.exists():
        old_df = pd.read_csv(raw_csv_path)
        merged_df = pd.concat([old_df, new_df], ignore_index=True)
        merged_df = merged_df.drop_duplicates(
            subset=["model_name", "scenario_id"],
            keep="last"
        )
    else:
        merged_df = new_df.copy()

    merged_df.to_csv(raw_csv_path, index=False, encoding="utf-8")
    return merged_df

if RUN_INFERENCE:
    if df_dynamic is None or df_dynamic.empty:
        raise ValueError("Сценарії відсутні або порожні.")

    SCENARIOS_LIST = df_dynamic.to_dict("records")

    existing_outputs_df, completed_pairs = load_existing_outputs_for_resume(RAW_OUTPUTS_CSV)

    if not existing_outputs_df.empty:
        print(f"♻️ Знайдено існуючі raw outputs: {len(existing_outputs_df)} рядків")
        print(f"♻️ Completed pairs для пропуску: {len(completed_pairs)}")

    all_new_outputs = []

    for adapter in adapters_info:
        try:
            print("\n" + "=" * 90)
            print(f"🚀 СТАРТ ІНФЕРЕНСУ: {adapter['model_name']}")

            task_scenarios_all = [s for s in SCENARIOS_LIST if s["task"] == adapter["task"]]
            if not task_scenarios_all:
                print(f"⚠️ Для task={adapter['task']} сценарії не знайдені.")
                continue

            if SKIP_COMPLETED_SCENARIOS:
                task_scenarios = [
                    s for s in task_scenarios_all
                    if (adapter["model_name"], s["scenario_id"]) not in completed_pairs
                ]
            else:
                task_scenarios = task_scenarios_all

            print(f"   Усього сценаріїв для task={adapter['task']}: {len(task_scenarios_all)}")
            print(f"   До запуску після skip completed: {len(task_scenarios)}")

            if not task_scenarios:
                print(f"✅ Усі сценарії для {adapter['model_name']} уже згенеровані. Пропускаю.")
                continue

            is_mock_adapter = bool(adapter.get("is_mock", False))
            tokenizer, model = (None, None)
            if not is_mock_adapter:
                tokenizer, model = load_model_for_inference(
                    adapter["base_model"],
                    adapter["adapter_dir"]
                )

            current_model_outputs = []

            for scen in tqdm(task_scenarios, desc=f"Генерація ({adapter['model_name']})"):
                max_tokens = MAX_NEW_TOKENS_BY_TASK.get(scen["task"], 180)
                prompt = build_inference_prompt(scen)
                if is_mock_adapter:
                    answer = generate_mock_answer(scen)
                else:
                    answer = generate_answer(tokenizer, model, prompt, max_new_tokens=max_tokens)
                pred_label = extract_label_from_output(answer) if scen["task"] == "disinfo" else None

                row = {
                    "model_name": adapter["model_name"],
                    "task": scen["task"],
                    "scenario_id": scen["scenario_id"],
                    "metric_category": scen["metric_category"],
                    "language": scen.get("language", FALLBACK_LANGUAGE),
                    "detected_language": scen.get("detected_language", scen.get("language", FALLBACK_LANGUAGE)),
                    "prompt_language": scen.get("prompt_language", scen.get("language", FALLBACK_LANGUAGE)),
                    "language_confidence": scen.get("language_confidence", 1.0),
                    "is_mixed_language": scen.get("is_mixed_language", 0),
                    "language_debug": scen.get("language_debug", ""),
                    "input_text": scen["input_text"],
                    "expert_ground_truth": scen["expert_ground_truth"],
                    "gold_label": scen.get("gold_label", None),
                    "expected_output_format": scen.get("expected_output_format", None),
                    "source_entity": scen.get("source_entity", None),
                    "model_answer": answer,
                    "predicted_label": pred_label,
                }

                current_model_outputs.append(row)
                all_new_outputs.append(row)

            model_outputs_df = pd.DataFrame(current_model_outputs)

            if not model_outputs_df.empty:
                merged_df = save_incremental_outputs(current_model_outputs, RAW_OUTPUTS_CSV)

                new_pairs = set(
                    zip(
                        model_outputs_df["model_name"].astype(str),
                        model_outputs_df["scenario_id"].astype(str)
                    )
                )
                completed_pairs.update(new_pairs)

                print(f"✅ Проміжні результати збережено після моделі {adapter['model_name']}")
                print(f"✅ Поточний розмір RAW_OUTPUTS_CSV: {len(merged_df)} рядків")

                if DISPLAY_PREVIEW_AFTER_EACH_MODEL:
                    display(HTML(f"<h3>Попередній перегляд відповідей: {adapter['model_name']}</h3>"))

                    summary_counts = (
                        model_outputs_df.groupby(["metric_category", "prompt_language"])
                        .size()
                        .reset_index(name="n")
                        .sort_values(["metric_category", "prompt_language"])
                    )
                    display(summary_counts)

                    preview_df = build_preview_table(model_outputs_df)
                    display(preview_df.head(PREVIEW_ROWS_PER_MODEL))

            cleanup_loaded_objects(model, tokenizer)

        except Exception as e:
            print(f"❌ ПОМИЛКА для {adapter['model_name']}: {e}")
            cleanup_loaded_objects()

    if RAW_OUTPUTS_CSV.exists():
        results_df = pd.read_csv(RAW_OUTPUTS_CSV)
        print(f"\n✅ Підсумковий файл raw outputs: {RAW_OUTPUTS_CSV}")
        print(f"✅ Усього рядків у raw outputs: {len(results_df)}")

        if not results_df.empty:
            print("\nМоделі, для яких уже є відповіді:")
            for x in sorted(results_df["model_name"].unique()):
                print(" -", x)

            display(results_df.groupby(["task", "model_name"]).size().reset_index(name="n"))

            review_df = results_df.copy()
            review_df["input_preview"] = review_df["input_text"].apply(lambda x: truncate_text(x, 180))
            review_df["answer_preview"] = review_df["model_answer"].apply(lambda x: truncate_text(x, 220))

            review_cols = [
                "model_name",
                "task",
                "metric_category",
                "language",
                "detected_language",
                "prompt_language",
                "scenario_id",
                "gold_label",
                "predicted_label",
                "expert_ground_truth",
                "input_preview",
                "answer_preview",
                "input_text",
                "model_answer",
            ]
            review_cols = [c for c in review_cols if c in review_df.columns]
            review_df = review_df[review_cols]

            display(HTML("<h3>Загальна таблиця для ручного перегляду відповідей</h3>"))
            display(review_df.head(20))

            # Завжди зберігаємо CSV-версію для перегляду + Excel (якщо встановлено openpyxl)
            review_csv = EVAL_DIR / "manual_review_outputs.csv"
            review_df.to_csv(review_csv, index=False, encoding="utf-8")
            print(f"✅ CSV для ручної перевірки збережено у: {review_csv}")

            if SAVE_EXCEL_REVIEW_FILE:
                review_xlsx = EVAL_DIR / "manual_review_outputs.xlsx"
                try:
                    with pd.ExcelWriter(review_xlsx, engine="openpyxl") as writer:
                        review_df.to_excel(writer, index=False, sheet_name="all_outputs")

                        summary_df = (
                            results_df.groupby(["model_name", "task", "metric_category", "prompt_language"])
                            .size()
                            .reset_index(name="n")
                            .sort_values(["model_name", "task", "metric_category", "prompt_language"])
                        )
                        summary_df.to_excel(writer, index=False, sheet_name="summary")

                    print(f"✅ Excel для ручної перевірки збережено у: {review_xlsx}")
                except Exception as excel_err:
                    print(f"⚠️ Не вдалося зберегти .xlsx ({excel_err}), але дані збережено у {review_csv}")

            # Також формуємо підсумковий файл оцінки FINAL_JUDGE_CSV для повної перевірки пайплайну
            judge_df = results_df.copy()
            judge_df["label_match"] = (
                (judge_df["gold_label"].astype(str) == judge_df["predicted_label"].astype(str)).astype(int)
                if "gold_label" in judge_df.columns and "predicted_label" in judge_df.columns
                else 1
            )
            judge_df["judge_safety_score"] = 5
            judge_df["judge_factuality_score"] = 5
            judge_df["judge_stability_score"] = 5
            judge_df.to_csv(FINAL_JUDGE_CSV, index=False, encoding="utf-8")
            print(f"✅ Підсумковий файл оцінки збережено у: {FINAL_JUDGE_CSV}")
        else:
            print("⚠️ RAW_OUTPUTS_CSV існує, але порожній.")
    else:
        print("\n⚠️ RAW_OUTPUTS_CSV не створено.")
